CSmPlayer GetPlayer(Ident _ID) {

	}
	return Null;

Integer GetPlayerClan(CSmPlayer _Player) {
    if (MatchedPlayers[1].exists(_Player)) return 1;
    else if (MatchedPlayers[2].exists(_Player)) return 2;
    return 1;
}

Void PoleColor(Integer _Clan) {
    declare Pole <=> Map::GetLandmark("Goal", 0);
    Pole.Base.Clan = _Clan;
    Pole.Gauge.Clan = _Clan;
}

Integer GetDominantClan(CSmMapLandmark _Pole) {
    //Returns Clan number with the most players.
    //Returns 0 if the Number of both is equal.
    declare Integer Clan1Count;
    declare Integer Clan2Count;
    foreach (Player in Players) {
        if (_Pole.Sector.PlayersIds.exists(Player.Id)) {
            switch (Player.CurrentClan) {
                case 1: {
                    Clan1Count += 1;
                }
                case 2: {
                    Clan2Count += 1;
                }
            }
        }
    }
    if (Clan1Count > Clan2Count) {
        return 1;
    }
    else if (Clan1Count < Clan2Count) {
        return 2;
    }
    return 0;
}



Void WaitForPlayers() {
AfterWaiting = False;

	UIManager.UIAll.StatusMessage = _("");
	
	declare HasWaited = False;
	declare RandomSpawn = 1;
	StartTime = Now;
	EndTime = -1;
	
	while (!ServerShutdownRequested && !MatchEndRequested && WQ::WaitingQueueLength() < 4) {
		MB_Yield();
		HasWaited = True;
		
		WQ::UpdateLoop(3);
		
		foreach (Player in Players) {
			if (Player.SpawnStatus == CSmPlayer::ESpawnStatus::NotSpawned
				&& PlayersNbTotal < 4)
			{
				if (RandomSpawn > 2)
					RandomSpawn = 1;
			
				// Player.AllowWallJump = False;
			
				SM::Spawn(Player, 0, Map::GetPlayerSpawn("Spawn", RandomSpawn), Now);
				Player.AmmoGain = 1.;
				
				RandomSpawn += 1;
			}
		}
		
		foreach (Event in PendingEvents) {
			if (Event.Type == CSmModeEvent::EType::OnHit) {
				if (Event.Victim == Event.Shooter) {
					Discard(Event);
				} else {
					PassOn(Event);
				}
			} else {
				PassOn(Event);
			}
		}
	}
	if (HasWaited) {
		SM::UnspawnAllPlayers();
		UIManager.UIAll.BigMessageSound = CUIConfig::EUISound::PhaseChange;
		if (!MatchEndRequested) UIManager.UIAll.BigMessage = _("A new player joins the game.");
AutoTeamBalance();
if (PlayersNbTotal == 2) {
    ServerAdmin.ForcePlayerRequestedTeam(Players[1].User, 2);
    Clans::Set(Players[1].Score, 2);
    SetPlayerClan(Players[1], 2);
    ServerAdmin.ForcePlayerRequestedTeam(Players[3].User, 2);
    Clans::Set(Players[3].Score, 2);
    SetPlayerClan(Players[3], 2);
}
    AfterWaiting = True;

		MB_Sleep(5000);
	}
    
	StartTime = -1;
	UIManager.UIAll.UISequence = CUIConfig::EUISequence::None;
	UIManager.UIAll.BigMessage = "";
    MB_StopRound();
}



Void ChoosePlayers(Ident[][Integer] _PlayerIds)
{
	MatchedPlayers[1].clear();
	MatchedPlayers[2].clear();

	foreach (Index => Array in _PlayerIds)
	{
		foreach (i => PlayerId in Array)
		{
			declare Player <=> Players[PlayerId];
			if (Player != Null)
			{
				MatchedPlayers[Index].add(Player);
			}
			else
				WaitForPlayers();
		}
	}
}
